21Catalog NarrowCatalog(const Catalog &catalog, int maxMagnitude, int maxStars, decimal minSeparation) {
22 Catalog result;
23 for (int i = 0; i < (int)catalog.size(); i++) {
24 // Higher magnitude implies the star is dimmer
25 if (catalog[i].magnitude <= maxMagnitude) {
26 result.push_back(catalog[i]);
27 }
28 }
29
30 std::sort(result.begin(), result.end(), CatalogStarMagnitudeCompare);
31
32 // remove stars that are too close to each other
33 std::set<int> tooCloseIndices;
34 // filter out stars that are too close together
35 // easy enough to n^2 brute force, the catalog isn't that big
36 for (int i = 0; i < (int)result.size(); i++) {
37 for (int j = i+1; j < (int)result.size(); j++) {
38 if (AngleUnit(result[i].spatial, result[j].spatial) < minSeparation) {
39 tooCloseIndices.insert(i);
40 tooCloseIndices.insert(j);
41 }
42 }
43 }
44
45 // Erase all the stars whose indices are in tooCloseIndices from the result.
46 // Loop backwards so indices don't get messed up as we iterate.
47 for (auto it = tooCloseIndices.rbegin(); it != tooCloseIndices.rend(); it++) {
48 result.erase(result.begin() + *it);
49 }
50
51 // and finally limit to n brightest stars
52 if (maxStars < (int)result.size()) {
53 result.resize(maxStars);
54 }
55
56 return result;
57}
