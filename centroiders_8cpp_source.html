80// a simple, but well tested thresholding algorithm that works well with star images
81int BasicThreshold(unsigned char *image, int imageWidth, int imageHeight,
82 std::vector<unsigned char> *rowMax = nullptr) {
83 unsigned long totalMag = 0;
84 decimal std = 0;
85 long totalPixels = imageHeight * imageWidth;
86 if (rowMax != nullptr) rowMax->assign(imageHeight, 0);
87 for (int y = 0; y < imageHeight; y++) {
88 const unsigned char *row = image + (long)y * imageWidth;
89 unsigned char brightest = 0;
90 for (int x = 0; x < imageWidth; x++) {
91 totalMag += row[x];
92 brightest = std::max(brightest, row[x]);
93 }
94 if (rowMax != nullptr) (*rowMax)[y] = brightest;
95 }
96 decimal mean = DECIMAL(totalMag) / totalPixels;
97 for (long i = 0; i < totalPixels; i++) {
98 std += DECIMAL_POW(image[i] - mean, 2);
99 }
100 std = DECIMAL_SQRT(std / totalPixels);
101 return mean + (std * 5);
102}

104// Is q + k <= mean + 5 * std, where mean = q + r/n?
105static bool WithinThreshold(uint64_t k, uint64_t n, uint64_t r, uint64_t scaledSquares) {
106 uint64_t added = scaledSquares + 2 * k * r;
107 uint64_t subtracted = k * k * n;
108 if (added < subtracted) return false;
109 uint64_t rest = added - subtracted;
110 if (rest >= 26 * n) return true;
111 return 26 * r * r <= n * rest;
112}

114// floor(mean + 5 * standard deviation) of all pixels.
115// exact for images up to 840,000,000 pixels
116// larger images use floating point.
117// If rowMax is given, it is filled with the brightest pixel of each row, like BasicThreshold.
118int BasicThresholdOnePass(unsigned char *image, int imageWidth, int imageHeight,
119 std::vector<unsigned char> *rowMax = nullptr) {
120 const uint64_t kMaxExactPixels = 840000000;
121 const int kChunk = 65536;
122
123 if (rowMax != nullptr) rowMax->assign(std::max(imageHeight, 0), 0);
124 if (imageWidth <= 0 || imageHeight <= 0) return 0;
125 uint64_t n = (uint64_t)imageWidth * imageHeight;
126
127 uint64_t sum = 0, sumSquares = 0;
128 for (int y = 0; y < imageHeight; y++) {
129 const unsigned char *row = image + (size_t)y * imageWidth;
130 unsigned char brightest = 0;
131 for (int x0 = 0; x0 < imageWidth;) {
132 int x1 = x0 + std::min(kChunk, imageWidth - x0);
133 uint32_t chunkSum = 0, chunkSquares = 0;
134 for (int x = x0; x < x1; x++) {
135 uint32_t pixel = row[x];
136 chunkSum += pixel;
137 chunkSquares += pixel * pixel;
138 brightest = std::max(brightest, row[x]);
139 }
140 sum += chunkSum;
141 sumSquares += chunkSquares;
142 x0 = x1;
143 }
144 if (rowMax != nullptr) (*rowMax)[y] = brightest;
145 }
146
147 if (n > kMaxExactPixels) {
148 double mean = (double)sum / n;
149 return (int)(mean + 5 * std::sqrt(std::max((double)sumSquares / n - mean * mean, 0.0)));
150 }
151
152 uint64_t q = sum / n, r = sum % n;
153 uint64_t scaledSquares = 25 * (sumSquares + q * q * n - 2 * q * sum);
154
155 uint64_t k = 0;
156 while (WithinThreshold(k + 1, n, r, scaledSquares)) k++;
157 return (int)(q + k);
158}

172std::vector<Star> CenterOfGravityAlgorithm::Go(unsigned char* image, int imageWidth, int imageHeight) const {
173 std::vector<Star> result;
174
175 std::vector<unsigned char> rowMax;
176 int cutoff = BasicThreshold(image, imageWidth, imageHeight, &rowMax);
177 const int offsets[] = {1, -1, imageWidth, -imageWidth};
178 const long n = imageWidth * imageHeight;
179 std::vector<bool> visited(n, false);
180
181 for (int row = 0; row < imageHeight; row++) {
182 if (rowMax[row] < cutoff) continue;
183 const long rowEnd = (long)(row + 1) * imageWidth;
184 for (long i = (long)row * imageWidth; i < rowEnd; i++) {
185 if (image[i] >= cutoff && !visited[i]) {
186 visited[i] = true;
187
188 // If any pixels are on the edge of the image then we don't centroid
189 // this star, in case part of the star is missing
190 bool isValid = true;
191 int centroidSize = 0;
192 std::vector<long> queue = {i};
193 CentroidParams p(i % imageWidth, i / imageWidth);
194
195 while (!queue.empty()) {
196 long ind = queue.back();
197 queue.pop_back();
198 centroidSize++;
199 long x = ind % imageWidth, y = ind / imageWidth;
200 if (x == 0 || x == imageWidth - 1 || y == 0 || y == imageHeight - 1) {
201 isValid = false;
202 }
203
204 p.magSum += image[ind];
205 p.xCoordMagSum += x * image[ind];
206 p.yCoordMagSum += y * image[ind];
207
208 if (x > p.xMax) p.xMax = x;
209 if (x < p.xMin) p.xMin = x;
210 if (y > p.yMax) p.yMax = y;
211 if (y < p.yMin) p.yMin = y;
212
213 for (int offset : offsets) {
214 long ni = ind + offset;
215 if (ni >= 0 && ni < n && std::abs((ni % imageWidth)-x) <= 1 && image[ni] >= cutoff && !visited[ni]) {
216 queue.push_back(ni);
217 visited[ni] = true;
218 }
219 }
220 }
221
222 int xDiameter = (p.xMax - p.xMin) + 1;
223 int yDiameter = (p.yMax - p.yMin) + 1;
224
225 if (isValid && p.magSum > 0) {
226 decimal xCoord = p.xCoordMagSum / DECIMAL(p.magSum);
227 decimal yCoord = p.yCoordMagSum / DECIMAL(p.magSum);
228 result.push_back(Star(xCoord + DECIMAL(0.5), yCoord + DECIMAL(0.5), (xDiameter)/DECIMAL(2.0), (yDiameter)/DECIMAL(2.0), centroidSize));
229 }
230 }
231 }
232 }
233 return result;
234}
