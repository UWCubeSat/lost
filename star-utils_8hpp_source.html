141// Generates all ways to choose k items from n, prioritizing patterns whose
142// dimmest star is as bright as possible (intuitively we want to avoid a bad centroid ruining a
143// pattern) E.g. for k=4, n=8: 0123, 0124, 0134, 0234, 1234, 0125, ... Formally, all combinations
144// using only items 0..m are exhausted before item m+1 appears
145class CombinationIterator {
146 public:
147 CombinationIterator(int k, int n) : k_(k), n_(n), indices_(k), isFirst_(true) {
148 for (int i = 0; i < k; i++)
149 indices_[i] = i;
150 }
151
152 bool next() {
153 if (isFirst_) {
154 isFirst_ = false;
155 return n_ >= k_;
156 }
157 int i = 0;
158 // Find leftmost index that can be incremented
159 while (i < k_ && indices_[i] + 1 == (i + 1 < k_ ? indices_[i + 1] : n_)) {
160 i++;
161 }
162 if (i == k_) return false;
163 indices_[i]++;
164 // Reset everything before the index we just changed
165 for (int j = 0; j < i; j++)
166 indices_[j] = j;
167 return true;
168 }
169
170 const std::vector<int>& indices() const { return indices_; }
171
172 private:
173 int k_, n_;
174 std::vector<int> indices_;
175 bool isFirst_;
176};

182// Check whether all pairs of stars have angular distance within maxFov.
183// SpatialAccessor(int i) must return const Vec3&.
184template <typename SpatialAccessor>
185bool AllPairsWithinFov(int numStars, SpatialAccessor spatial, decimal cosMaxFov) {
186 for (int i = 0; i < numStars; i++) {
187 for (int j = i + 1; j < numStars; j++) {
188 if (spatial(i) * spatial(j) <= cosMaxFov) return false;
189 }
190 }
191 return true;
192}

194// Compute sorted edge ratios for a star pattern. SpatialAccessor(int i) must return const Vec3&.
195template <typename SpatialAccessor>
196std::vector<decimal> TetraEdgeRatios(int numStars, SpatialAccessor spatial) {
197 std::vector<decimal> edgeLengths;
198 for (int i = 0; i < numStars; i++) {
199 for (int j = i + 1; j < numStars; j++) {
200 edgeLengths.push_back((spatial(j) - spatial(i)).Magnitude());
201 }
202 }
203 std::sort(edgeLengths.begin(), edgeLengths.end());
204 decimal longestEdgeLen = edgeLengths.back();
205 std::vector<decimal> edgeRatios;
206 for (size_t i = 0; i < edgeLengths.size() - 1; i++) {
207 edgeRatios.push_back(edgeLengths[i] / longestEdgeLen);
208 }
209 return edgeRatios;
210}
