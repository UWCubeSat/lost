136// Generates all ways to choose k items from n, prioritizing patterns whose
137// dimmest star is as bright as possible (intuitively we want to avoid a bad centroid ruining a
138// pattern) E.g. for k=4, n=8: 0123, 0124, 0134, 0234, 1234, 0125, ... Formally, all combinations
139// using only items 0..m are exhausted before item m+1 appears
140class CombinationIterator {
141 public:
142 CombinationIterator(int k, int n) : k_(k), n_(n), indices_(k), isFirst_(true) {
143 for (int i = 0; i < k; i++)
144 indices_[i] = i;
145 }
146
147 bool next() {
148 if (isFirst_) {
149 isFirst_ = false;
150 return n_ >= k_;
151 }
152 int i = 0;
153 // Find leftmost index that can be incremented
154 while (i < k_ && indices_[i] + 1 == (i + 1 < k_ ? indices_[i + 1] : n_)) {
155 i++;
156 }
157 if (i == k_) return false;
158 indices_[i]++;
159 // Reset everything before the index we just changed
160 for (int j = 0; j < i; j++)
161 indices_[j] = j;
162 return true;
163 }
164
165 const std::vector<int>& indices() const { return indices_; }
166
167 private:
168 int k_, n_;
169 std::vector<int> indices_;
170 bool isFirst_;
171};

177// Check whether all pairs of stars have angular distance within maxFov.
178// SpatialAccessor(int i) must return const Vec3&.
179template <typename SpatialAccessor>
180bool AllPairsWithinFov(int numStars, SpatialAccessor spatial, decimal cosMaxFov) {
181 for (int i = 0; i < numStars; i++) {
182 for (int j = i + 1; j < numStars; j++) {
183 if (spatial(i) * spatial(j) <= cosMaxFov) return false;

189// Compute sorted edge ratios for a star pattern. SpatialAccessor(int i) must return const Vec3&.
190template <typename SpatialAccessor>
191std::vector<decimal> TetraEdgeRatios(int numStars, SpatialAccessor spatial) {
192 std::vector<decimal> edgeLengths;
193 for (int i = 0; i < numStars; i++) {
194 for (int j = i + 1; j < numStars; j++) {
195 edgeLengths.push_back((spatial(j) - spatial(i)).Magnitude());
196 }
197 }
198 std::sort(edgeLengths.begin(), edgeLengths.end());
199 decimal longestEdgeLen = edgeLengths.back();
200 std::vector<decimal> edgeRatios;
201 for (size_t i = 0; i < edgeLengths.size() - 1; i++) {
202 edgeRatios.push_back(edgeLengths[i] / longestEdgeLen);
203 }
204 return edgeRatios;
205}
