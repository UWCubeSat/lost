62 FILE *file;
63 decimal raj2000, dej2000;
64 int magnitudeHigh, magnitudeLow, name;
65 char weird;
66
67 file = fopen(tsvPath.c_str(), "r");
68
69 if (file == NULL) {
70 printf("Error opening file: %s\n", strerror(errno));
71 exit(1); // TODO: do we want any other error handling?
72 return result;
73 }
74
75 #ifdef LOST_FLOAT_MODE
76 std::string format = "%f|%f|%d|%c|%d.%d";
77 #else
78 std::string format = "%lf|%lf|%d|%c|%d.%d";
79 #endif
80
81 while (EOF != fscanf(file, format.c_str(),
82 &raj2000, &dej2000,
83 &name, &weird,
84 &magnitudeHigh, &magnitudeLow)) {
85 result.push_back(CatalogStar(DegToRad(raj2000),

89 }

101const Catalog &CatalogRead() {
102 static bool readYet = false;
103 static std::vector<CatalogStar> catalog;
104
105 if (!readYet) {
106 readYet = true;
107 char *tsvPath = getenv("LOST_BSC_PATH");
108 catalog = BscParse(tsvPath ? tsvPath : DEFAULT_BSC_PATH);
109 // perform essential narrowing
110 // remove all stars with exactly the same position as another, keeping the one with brighter magnitude
111 std::sort(catalog.begin(), catalog.end(), [](const CatalogStar &a, const CatalogStar &b) {
112 return a.spatial.x < b.spatial.x;
113 });
114 for (int i = catalog.size()-1; i > 0; i--) {
115 if ((catalog[i].spatial - catalog[i-1].spatial).Magnitude() < DECIMAL(5e-5)) { // 70 stars removed at this threshold.
116 if (catalog[i].magnitude > catalog[i-1].magnitude) {
117 catalog.erase(catalog.begin() + i);
118 } else {
119 catalog.erase(catalog.begin() + i - 1);
120 }
121 }
122 }
123 }
124 return catalog;
125}

129unsigned char *SurfaceToGrayscaleImage(cairo_surface_t *cairoSurface) {
130 int width, height;
131 unsigned char *result;
132 uint32_t *cairoImage, pixel;
133
134 if (cairo_image_surface_get_format(cairoSurface) != CAIRO_FORMAT_ARGB32 &&
135 cairo_image_surface_get_format(cairoSurface) != CAIRO_FORMAT_RGB24) {
136 puts("Can't convert weird image formats to grayscale.");
137 return NULL;
138 }
139
140 width = cairo_image_surface_get_width(cairoSurface);
141 height = cairo_image_surface_get_height(cairoSurface);
142
143 result = new unsigned char[width*height];
144 cairoImage = (uint32_t *)cairo_image_surface_get_data(cairoSurface);
145
146 for (int i = 0; i < height * width; i++) {
147 pixel = cairoImage[i];
148 // use "luminosity" method of grayscaling
149 result[i] = round(
150 (pixel>>16 &0xFF) *0.21 +
151 (pixel>>8 &0xFF) *0.71 +
152 (pixel &0xFF) *0.07);
153 }
154
155 return result;
156}

158cairo_surface_t *GrayscaleImageToSurface(const unsigned char *image,
159 const int width, const int height) {
160 // cairo's 8-bit type isn't BW, it's an alpha channel only, which would look a bit weird lol
161 cairo_surface_t *result = cairo_image_surface_create(CAIRO_FORMAT_RGB24, width, height);
162 uint32_t *resultData = (uint32_t *)cairo_image_surface_get_data(result);
163 // hopefully unnecessary
164 cairo_surface_flush(result);
165 for (long i = 0; i < width * height; i++) {
166 // equal r, g, and b components
167 resultData[i] = (image[i] << 16) + (image[i] << 8) + (image[i]);
168 }
169 cairo_surface_mark_dirty(result);
170 return result;
171}

469public:
470 GeneratedStar(Star star, decimal peakBrightness, Vec2 motionBlurDelta)
471 : Star(star), peakBrightness(peakBrightness), delta(motionBlurDelta) { };
472
474 decimal peakBrightness;
475
477 Vec2 delta;

480// In the equations for pixel brightness both with motion blur enabled and disabled, we don't need
481// any constant scaling factor outside the integral because when d0=0, the brightness at the center
482// will be zero without any scaling. The scaling factor you usually see on a Normal distribution is
483// so that the Normal distribution integrates to one over the real line, making it a probability
484// distribution. But we want the /peak/ to be one, not the integral. motion blur enabled
485
496static decimal MotionBlurredPixelBrightness(const Vec2 &pixel, const GeneratedStar &generatedStar,
497 decimal t, decimal stddev) {
498 const Vec2 &p0 = generatedStar.position;
499 const Vec2 &delta = generatedStar.delta;
500 const Vec2 d0 = p0 - pixel;
501 return generatedStar.peakBrightness
502 * stddev*DECIMAL_SQRT(DECIMAL_M_PI) / (DECIMAL_SQRT(2)*delta.Magnitude())
503 * DECIMAL_EXP(DECIMAL_POW(d0.x*delta.x + d0.y*delta.y, 2) / (2*stddev*stddev*delta.MagnitudeSq())
504 - d0.MagnitudeSq() / (2*stddev*stddev))
505 * DECIMAL_ERF((t*delta.MagnitudeSq() + d0.x*delta.x + d0.y*delta.y) / (stddev*DECIMAL_SQRT(2)*delta.Magnitude()));
506}

509static decimal StaticPixelBrightness(const Vec2 &pixel, const GeneratedStar &generatedStar,
510 decimal t, decimal stddev) {
511 const Vec2 d0 = generatedStar.position - pixel;
512 return generatedStar.peakBrightness * t * DECIMAL_EXP(-d0.MagnitudeSq() / (2 * stddev * stddev));
513}

524static decimal CentroidImagingProbability(decimal mag, decimal cutoffMag) {
525 decimal brightness = MagToBrightness(mag);
526 decimal cutoffBrightness = MagToBrightness(cutoffMag);
527 decimal stddev = cutoffBrightness/DECIMAL(5.0);
528 // CDF of Normal distribution with given mean and stddev
529 return 1 - (DECIMAL(0.5) * (1 + DECIMAL_ERF((cutoffBrightness-brightness)/(stddev*DECIMAL_SQRT(2.0)))));
530}

538GeneratedPipelineInput::GeneratedPipelineInput(const Catalog &catalog,
539 Attitude attitude,
540 Camera camera,
541 std::default_random_engine *rng,
542
543 bool centroidsOnly,
544 decimal zeroMagTotalPhotons,
545 decimal starSpreadStdDev,
546 decimal saturationPhotons,
547 decimal darkCurrent,
548 decimal readNoiseStdDev,
549 Attitude motionBlurDirection, // applied on top of the attitude
550 decimal exposureTime,
551 decimal readoutTime, // zero for no rolling shutter
552 bool shotNoise,
553 int oversampling,
554 int numFalseStars,
555 int falseStarMinMagnitude,
556 int falseStarMaxMagnitude,
557 int cutoffMag,
558 decimal perturbationStddev)
559 : camera(camera), attitude(attitude), catalog(catalog), centroidsOnly(centroidsOnly) {
560
561 assert(falseStarMaxMagnitude <= falseStarMinMagnitude);
562 assert(perturbationStddev >= DECIMAL(0.0));
563
564 image.width = camera.XResolution();
565 image.height = camera.YResolution();
566 // number of true photons each pixel receives.
567
568 assert(oversampling >= 1);
569 int oversamplingPerAxis = DECIMAL_CEIL(DECIMAL_SQRT(oversampling));
570 if (oversamplingPerAxis*oversamplingPerAxis != oversampling) {
571 std::cerr << "WARNING: oversampling was not a perfect square. Rounding up to "
572 << oversamplingPerAxis*oversamplingPerAxis << "." << std::endl;
573 }
574 assert(exposureTime > 0);
575 bool motionBlurEnabled = abs(motionBlurDirection.GetQuaternion().Angle()) > DECIMAL(0.001);
576 Quaternion motionBlurDirectionQ = motionBlurDirection.GetQuaternion();
577 // attitude at the middle of exposure time
578 Quaternion currentAttitude = attitude.GetQuaternion();
579 // attitude 1 time unit after middle of exposure
580 Quaternion futureAttitude = motionBlurDirectionQ*currentAttitude;
581 std::vector<GeneratedStar> generatedStars;
582
583 // a star with 1 photon has peak density 1/(2pi sigma^2), because 2d gaussian formula. Then just
584 // multiply up proportionally!
585 decimal zeroMagPeakPhotonDensity = zeroMagTotalPhotons / (2*DECIMAL_M_PI * starSpreadStdDev*starSpreadStdDev);
586
587 Catalog catalogWithFalse = catalog;
588
589 std::uniform_real_distribution<decimal> uniformDistribution(DECIMAL(0.0), DECIMAL(1.0));
590 std::uniform_int_distribution<int> magnitudeDistribution(falseStarMaxMagnitude, falseStarMinMagnitude);
591 for (int i = 0; i < numFalseStars; i++) {
592 decimal ra = uniformDistribution(*rng) * 2*DECIMAL_M_PI;
593 // to be uniform around sphere. Borel-Kolmogorov paradox is calling
594 decimal de = DECIMAL_ASIN(uniformDistribution(*rng)*2 - 1);
595 decimal magnitude = magnitudeDistribution(*rng);
596
597 catalogWithFalse.push_back(CatalogStar(ra, de, magnitude, -1));
598 }
599
600 for (int i = 0; i < (int)catalogWithFalse.size(); i++) {
601 bool isTrueStar = i < (int)catalog.size();
602
603 const CatalogStar &catalogStar = catalogWithFalse[i];
604 Vec3 rotated = attitude.Rotate(catalogWithFalse[i].spatial);
605 if (rotated.x <= 0) {
606 continue;
607 }
608 Vec2 camCoords = camera.SpatialToCamera(rotated);
609
610 if (camera.InSensor(camCoords)) {
611 Vec3 futureSpatial = futureAttitude.Rotate(catalogWithFalse[i].spatial);
612 Vec2 delta = camera.SpatialToCamera(futureSpatial) - camCoords;
613 if (!motionBlurEnabled) {
614 delta = {0, 0}; // avoid decimaling point funny business
615 }
616 // radiant intensity, in photons per time unit per pixel, at the center of the star.
617 decimal peakBrightnessPerTime = zeroMagPeakPhotonDensity * MagToBrightness(catalogStar.magnitude);
618 decimal interestingThreshold = DECIMAL(0.05); // we don't need to check pixels that are expected to
619 // receive this many photons or fewer.
620 // inverse of the function defining the Gaussian distribution: Find out how far from the
621 // mean we'll have to go until the number of photons is less than interestingThreshold
622 decimal radius = DECIMAL_CEIL(DECIMAL_SQRT(-DECIMAL_LOG(interestingThreshold/peakBrightnessPerTime/exposureTime)*2*DECIMAL_M_PI*starSpreadStdDev*starSpreadStdDev));
623 Star star = Star(camCoords.x, camCoords.y,
624 radius, radius,
625 // important to invert magnitude here, so that centroid magnitude becomes larger for brighter stars.
626 // It's possible to make it so that the magnitude is always positive too, but allowing weirder magnitudes helps keep star-id algos honest about their assumptions on magnitude.
627 // we don't use its magnitude anywhere else in generation; peakBrightness was already calculated.
628 -catalogStar.magnitude);
629 generatedStars.push_back(GeneratedStar(star, peakBrightnessPerTime, delta));
630
631 // Now add the star to the input and expected lists.
632 // We do actually want to add false stars as well, because:
633 // a) A centroider isn't any worse because it picks up a false star that looks exactly like a normal star, so why should we exclude them from compare-centroids?
634 // b) We want to feed false stars into star-ids to evaluate their false-star resilience without running centroiding.
635
636 // Add all stars to expected, cause that's how we roll
637 expectedStars.push_back(star);
638 // and provide expected identifications for all of them
639 if (isTrueStar) {
640 expectedStarIds.push_back(StarIdentifier(expectedStars.size()-1, i));
641 }
642
643 // for input, though, add perturbation and stuff.
644 Star inputStar = star;
645 if (perturbationStddev > DECIMAL(0.0)) {
646 std::normal_distribution<decimal> perturbation1DDistribution(DECIMAL(0.0), perturbationStddev);
647 // clamp to within 2 standard deviations for some reason:
648 inputStar.position.x += std::max(std::min(perturbation1DDistribution(*rng), 2*perturbationStddev), -2*perturbationStddev);
649 inputStar.position.y += std::max(std::min(perturbation1DDistribution(*rng), 2*perturbationStddev), -2*perturbationStddev);
650 }
651 // If it got perturbed outside of the sensor, don't add it.
652 if (camera.InSensor(inputStar.position)
653 // and also don't add it if it's too dim.
654 && (cutoffMag >= 10000 // but always add the star if the cutoff is very high
655 || !isTrueStar // and always add the false stars
656 || std::bernoulli_distribution((double) CentroidImagingProbability(catalogStar.magnitude, cutoffMag))(*rng))) {
657 inputStars.push_back(inputStar);
658 if (isTrueStar) {
659 inputStarIds.push_back(StarIdentifier(inputStars.size()-1, i));
660 }
661 }
662 }
663 }
664
665 if (centroidsOnly) {
666 // we're outta here
667 return;
668 }
669
670 std::vector<decimal> photonsBuffer(image.width*image.height, 0);
671
672 for (const GeneratedStar &star : generatedStars) {
673 // delta will be exactly (0,0) when motion blur disabled
674 Vec2 earliestPosition = star.position - star.delta*(exposureTime/DECIMAL(2.0) + readoutTime/DECIMAL(2.0));
675 Vec2 latestPosition = star.position + star.delta*(exposureTime/DECIMAL(2.0) + readoutTime/DECIMAL(2.0));
676 int xMin = std::max(0, (int)std::min(earliestPosition.x - star.radiusX, latestPosition.x - star.radiusX));
677 int xMax = std::min(image.width-1, (int)std::max(earliestPosition.x + star.radiusX, latestPosition.x + star.radiusX));
678 int yMin = std::max(0, (int)std::min(earliestPosition.y - star.radiusX, latestPosition.y - star.radiusX));
679 int yMax = std::min(image.height-1, (int)std::max(earliestPosition.y + star.radiusX, latestPosition.y + star.radiusX));
680
681 // peak brightness is measured in photons per time unit per pixel, so if oversampling, we
682 // need to convert units to photons per time unit per sample
683 decimal oversamplingBrightnessFactor = oversamplingPerAxis*oversamplingPerAxis;
684
685 // the star.x and star.y refer to the pixel whose top left corner the star should appear at
686 // (and fractional amounts are relative to the corner). When we color a pixel, we ideally
687 // would integrate the intensity of the star over that pixel, but we can make do by sampling
688 // the intensity of the star at the /center/ of the pixel, ie, star.x+.5 and star.y+.5
689 for (int xPixel = xMin; xPixel <= xMax; xPixel++) {
690 for (int yPixel = yMin; yPixel <= yMax; yPixel++) {
691 // offset of beginning & end of readout compared to beginning & end of readout for
692 // center row
693 decimal readoutOffset = readoutTime * (yPixel - image.height/DECIMAL(2.0)) / image.height;
694 decimal tStart = -exposureTime/DECIMAL(2.0) + readoutOffset;
695 decimal tEnd = exposureTime/DECIMAL(2.0) + readoutOffset;
696
697 // loop through all samples in the current pixel
698 for (int xSample = 0; xSample < oversamplingPerAxis; xSample++) {
699 for (int ySample = 0; ySample < oversamplingPerAxis; ySample++) {
700 decimal x = xPixel + (xSample+DECIMAL(0.5))/oversamplingPerAxis;
701 decimal y = yPixel + (ySample+DECIMAL(0.5))/oversamplingPerAxis;
702
703 decimal curPhotons;
704 if (motionBlurEnabled) {
705 curPhotons =
706 (MotionBlurredPixelBrightness({x, y}, star, tEnd, starSpreadStdDev)
707 - MotionBlurredPixelBrightness({x, y}, star, tStart, starSpreadStdDev))
708 / oversamplingBrightnessFactor;
709 } else {
710 curPhotons = StaticPixelBrightness({x, y}, star, exposureTime, starSpreadStdDev)
711 / oversamplingBrightnessFactor;
712 }
713
714 assert(DECIMAL(0.0) <= curPhotons);
715
716 photonsBuffer[xPixel + yPixel*image.width] += curPhotons;
717 }
718 }
719 }
720 }
721 }
722
723 std::normal_distribution<decimal> standardNormalDist(DECIMAL(0.0), DECIMAL(1.0));
724
725 // convert from photon counts to observed pixel brightnesses, applying noise and such.
726 imageData = std::vector<unsigned char>(image.width*image.height);
727 image.image = imageData.data();
728 for (int i = 0; i < image.width * image.height; i++) {
729 decimal curBrightness = 0;
730
731 // dark current (Constant)
732 curBrightness += darkCurrent;
733
734 // read noise (Gaussian)
735 curBrightness += readNoiseStdDev * standardNormalDist(*rng);
736
737 // shot noise (Poisson), and quantize
738 long quantizedPhotons;
739 if (shotNoise) {
740 // with GNU libstdc++, it keeps sampling from the distribution until it's within the min-max
741 // range. This is problematic if the mean is far above the max long value, because then it
742 // might have to sample many many times (and furthermore, the results won't be useful
743 // anyway)
744 decimal photons = photonsBuffer[i];
745 if (photons > DECIMAL(LONG_MAX) - DECIMAL(3.0) * DECIMAL_SQRT(LONG_MAX)) {
746 std::cout << "ERROR: One of the pixels had too many photons. Generated image would not be physically accurate, exiting." << std::endl;
747 exit(1);
748 }
749 // std::poisson_distribution requires mean > 0
750 if (photons > DECIMAL(0.0)) {
751 std::poisson_distribution<long> shotNoiseDist((double) photons);
752 quantizedPhotons = shotNoiseDist(*rng);
753 } else {
754 quantizedPhotons = 0;
755 }
756 } else {
757 quantizedPhotons = round(photonsBuffer[i]);
758 }
759 curBrightness += quantizedPhotons / saturationPhotons;
760
761 // std::clamp not introduced until C++17, so we avoid it.
762 decimal clampedBrightness = std::max(std::min(curBrightness, DECIMAL(1.0)), DECIMAL(0.0));
763 imageData[i] = floor(clampedBrightness * kMaxBrightness); // TODO: off-by-one, 256?
764 }
765}

771static Attitude RandomAttitude(std::default_random_engine* pReng) {
772 std::uniform_real_distribution<decimal> randomAngleDistribution(0, 1);
773
774 // normally the ranges of the Ra and Dec are:
775 // Dec: [-90 deg, 90 deg] --> [-pi/2 rad, pi/2 rad], where negative means south
776 // of the celestial equator and positive means north
777 // Ra: [0 deg, 360 deg] --> [0 rad, 2pi rad ]
778 // Roll: [0 rad, 2 pi rad]
779
780 decimal randomRa = 2 * DECIMAL_M_PI * randomAngleDistribution(*pReng);
781 decimal randomDec = (DECIMAL_M_PI / 2) - acos(1 - 2 * randomAngleDistribution(*pReng)); //acos returns a decimal in range [0, pi]
782 decimal randomRoll = 2 * DECIMAL_M_PI * randomAngleDistribution(*pReng);
783
784 Attitude randAttitude = Attitude(SphericalToQuaternion(randomRa, randomDec, randomRoll));
785
786 return randAttitude;
787}

983PipelineOutput Pipeline::Go(const PipelineInput &input) {
984 // Start executing the pipeline at the first stage that has both input and an algorithm. From
985 // there, execute each successive stage of the pipeline using the output of the last stage
986 // (human centipede) until there are no more stages set.
987 PipelineOutput result;
988
989 const Image *inputImage = input.InputImage();
990 const Stars *inputStars = input.InputStars();
991 const StarIdentifiers *inputStarIds = input.InputStarIds();
992
993 // if database is provided, that's where we get catalog from.
994 if (database) {
995 MultiDatabase multiDatabase(database.get());
996 const unsigned char *catalogBuffer = multiDatabase.SubDatabasePointer(kCatalogMagicValue);
997 if (catalogBuffer != NULL) {
998 DeserializeContext des(catalogBuffer);
999 result.catalog = DeserializeCatalog(&des, NULL, NULL);
1000 } else {
1001 std::cerr << "WARNING: That database does not include a catalog. Proceeding with the full catalog." << std::endl;
1002 result.catalog = input.GetCatalog();
1003 }
1004 } else {
1005 result.catalog = input.GetCatalog();
1006 }
1007
1008 if (centroidAlgorithm && inputImage) {
1009
1010 // run centroiding, keeping track of the time it takes
1011 std::chrono::time_point<std::chrono::steady_clock> start = std::chrono::steady_clock::now();
1012
1013 // TODO: we should probably modify Go to just take an image argument
1014 Stars unfilteredStars = centroidAlgorithm->Go(inputImage->image, inputImage->width, inputImage->height);
1015
1016 std::chrono::time_point<std::chrono::steady_clock> end = std::chrono::steady_clock::now();
1017 result.centroidingTimeNs = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
1018
1019 // MAGNITUDE FILTERING
1020 int minMagnitude = centroidMinMagnitude;
1021 if (centroidMinStars > 0
1022 // don't need to filter if we don't even have that many stars
1023 && centroidMinStars < (int)unfilteredStars.size()) {
1024
1025 Stars magSortedStars = unfilteredStars;
1026 // sort descending
1027 std::sort(magSortedStars.begin(), magSortedStars.end(), [](const Star &a, const Star &b) { return a.magnitude > b.magnitude; });
1028 minMagnitude = std::max(minMagnitude, magSortedStars[centroidMinStars - 1].magnitude);
1029 }
1030 // determine the minimum magnitude according to sorted stars
1031 Stars *filteredStars = new std::vector<Star>();
1032 for (const Star &star : unfilteredStars) {
1033 assert(star.magnitude >= 0); // catalog stars can have negative magnitude, but by our
1034 // conventions, centroids shouldn't.
1035 if (star.magnitude >= minMagnitude) {
1036 filteredStars->push_back(star);
1037 }
1038 }
1039 result.stars = std::unique_ptr<Stars>(filteredStars);
1040 inputStars = filteredStars;
1041
1042 // any starid set up to this point needs to be discarded, because it's based on input
1043 // centroids instead of our new centroids.
1044 inputStarIds = NULL;
1045 result.starIds = NULL;
1046 } else if (centroidAlgorithm) {
1047 std::cerr << "ERROR: Centroid algorithm specified, but no input image to run it on." << std::endl;
1048 exit(1);
1049 }
1050
1051 if (starIdAlgorithm && database && inputStars && input.InputCamera()) {
1052 // TODO: don't copy the vector!
1053 std::chrono::time_point<std::chrono::steady_clock> start = std::chrono::steady_clock::now();
1054
1055 result.starIds = std::unique_ptr<StarIdentifiers>(new std::vector<StarIdentifier>(
1056 starIdAlgorithm->Go(database.get(), *inputStars, result.catalog, *input.InputCamera())));
1057
1058 std::chrono::time_point<std::chrono::steady_clock> end = std::chrono::steady_clock::now();
1059 result.starIdTimeNs = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
1060
1061 inputStarIds = result.starIds.get();
1062 } else if (starIdAlgorithm) {
1063 std::cerr << "ERROR: Star ID algorithm specified but cannot run because database, centroids, or camera are missing." << std::endl;
1064 exit(1);
1065 }
1066
1067 if (attitudeEstimationAlgorithm && inputStarIds && input.InputCamera()) {
1068 assert(inputStars); // ensure that starIds doesn't exist without stars
1069 std::chrono::time_point<std::chrono::steady_clock> start = std::chrono::steady_clock::now();
1070
1071 result.attitude = std::unique_ptr<Attitude>(
1072 new Attitude(attitudeEstimationAlgorithm->Go(*input.InputCamera(), *inputStars, result.catalog, *inputStarIds)));
1073
1074 std::chrono::time_point<std::chrono::steady_clock> end = std::chrono::steady_clock::now();
1075 result.attitudeEstimationTimeNs = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
1076 } else if (attitudeEstimationAlgorithm) {
1077 std::cerr << "ERROR: Attitude estimation algorithm set, but either star IDs or camera are missing. One reason this can happen: Setting a centroid algorithm and attitude algorithm, but no star-id algorithm -- that can't work because the input star-ids won't properly correspond to the output centroids!" << std::endl;
1078 exit(1);
1079 }
1080
1081 return result;
1082}

1131static std::multimap<int, int> FindClosestCentroids(decimal threshold,
1132 const Stars &one,
1133 const Stars &two) {
1134 std::multimap<int, int> result;
1135
1136 for (int i = 0; i < (int)one.size(); i++) {
1137 std::vector<std::pair<decimal, int>> closest;
1138 for (int k = 0; k < (int)two.size(); k++) {
1139 decimal currDistance = (one[i].position - two[k].position).Magnitude();
1140 if (currDistance <= threshold) {
1141 closest.emplace_back(currDistance, k);
1142 }
1143 }
1144 std::sort(closest.begin(), closest.end());
1145 for (const std::pair<decimal, int> &pair : closest) {
1146 result.emplace(i, pair.second);
1147 }
1148 }

1150 return result;

1158CentroidComparison CentroidsCompare(decimal threshold,
1159 const Stars &expected,
1160 const Stars &actual) {
1161
1162 // TODO: Somehow penalize when multiple centroids correspond to the same expected star (i.e.,
1163 // one star turned into multiple centroids). That should probably be considered an extra
1164 // centroid, but rn it isn't.
1165
1166 CentroidComparison result;
1167 // maps from indexes in each list to the closest centroid from other list
1168 std::multimap<int, int> actualToExpected = FindClosestCentroids(threshold, actual, expected);
1169
1170 for (int i = 0; i < (int)actualToExpected.size(); i++) {
1171 auto closest = actualToExpected.find(i);
1172 if (closest == actualToExpected.end()) {
1173 result.numExtraCentroids++;
1174 } else {
1175 result.meanError += (actual[i].position - expected[closest->second].position).Magnitude();
1176 result.numCorrectCentroids++;
1177 }
1178 }
1179 result.meanError /= result.numCorrectCentroids;
1180
1181 return result;
1182}

1203StarIdComparison StarIdsCompare(const StarIdentifiers &expected, const StarIdentifiers &actual,
1204 // use these to map indices to names for the respective lists of StarIdentifiers
1205 const Catalog &expectedCatalog, const Catalog &actualCatalog,
1206 decimal centroidThreshold,
1207 const Stars &expectedStars, const Stars &inputStars) {

1209 StarIdComparison result = {
1210 0, // correct
1211 0, // incorrect
1212 0, // total
1213 };
1214
1215 // EXPECTED STAR IDS
1216
1217 // map from expected star indices to expected catalog indices (basically flattening the expected star-ids)
1218 std::vector<int> expectedCatalogIndices(expectedStars.size(), -1);
1219 for (const StarIdentifier &starId : expected) {
1220 assert(0 <= starId.starIndex && starId.starIndex <= (int)expectedStars.size());
1221 assert(0 <= starId.catalogIndex && starId.catalogIndex <= (int)expectedCatalog.size());
1222 expectedCatalogIndices[starId.starIndex] = starId.catalogIndex;
1223 }
1224
1225 // FIND NEAREST CENTROIDS
1226
1227 std::multimap<int, int> inputToExpectedCentroids = FindClosestCentroids(centroidThreshold, inputStars, expectedStars);
1228 // std::multimap<int, int> expectedToInputCentroids = FindClosestCentroids(centroidThreshold, expectedStars, inputStars);
1229
1230 // COMPUTE TOTAL
1231 // Count the number of expected stars with at least one input star near them
1232 for (int i = 0; i < (int)inputStars.size(); i++) {
1233 // make sure there's at least one expected star near this input star which has an identification
1234 auto closestRange = inputToExpectedCentroids.equal_range(i);
1235 bool found = false;
1236 for (auto it = closestRange.first; it != closestRange.second; it++) {
1237 if (expectedCatalogIndices[it->second] != -1) {
1238 found = true;
1239 break;
1240 }
1241 }
1242 if (found) {
1243 result.numTotal++;
1244 }
1245 }
1246
1247 // COMPUTE CORRECT AND INCORRECT
1248
1249 std::vector<bool> identifiedInputCentroids(inputStars.size(), false);
1250 for (const StarIdentifier &starId : actual) {
1251 // as later, there shouldn't be duplicate starIndex. This indicates a bug in the star-id algorithm, not comparison code.
1252 assert(!identifiedInputCentroids[starId.starIndex]);
1253 identifiedInputCentroids[starId.starIndex] = true;
1254 assert(0 <= starId.starIndex && starId.starIndex <= (int)inputStars.size());
1255 assert(0 <= starId.catalogIndex && starId.catalogIndex <= (int)actualCatalog.size());
1256
1257 // Check that there's at least one expected centroid in range which agrees with your identification.
1258 auto expectedCentroidsInRange = inputToExpectedCentroids.equal_range(starId.starIndex);
1259 bool found = false;
1260 for (auto it = expectedCentroidsInRange.first; it != expectedCentroidsInRange.second; it++) {

1262 if (expectedCatalogIndex != -1
1263 && expectedCatalog[expectedCatalogIndex].name == actualCatalog[starId.catalogIndex].name) {

1606static void PipelineComparatorAttitude(std::ostream &os,
1607 const PipelineInputList &expected,
1608 const std::vector<PipelineOutput> &actual,
1609 const PipelineOptions &values) {
1610
1611 // TODO: use Wahba loss function (maybe average per star) instead of just angle. Also break
1612 // apart roll error from boresight error. This is just quick and dirty for testing
1613
1614 decimal angleThreshold = DegToRad(values.attitudeCompareThreshold);
1615
1616 decimal attitudeErrorSum = 0.0f;
1617 int numCorrect = 0;
1618 int numIncorrect = 0;
1619
1620 for (int i = 0; i < (int)expected.size(); i++) {
1621 if (actual[i].attitude->IsKnown()) {
1622 Quaternion expectedQuaternion = expected[i]->ExpectedAttitude()->GetQuaternion();
1623 Quaternion actualQuaternion = actual[i].attitude->GetQuaternion();
1624 decimal attitudeError = (expectedQuaternion * actualQuaternion.Conjugate()).SmallestAngle();
1625 assert(attitudeError >= 0);
1626
1627 if (attitudeError <= angleThreshold) {
1628 attitudeErrorSum += attitudeError;
1629 numCorrect++;
1630 } else {
1631 numIncorrect++;
1632 }
1633 }
1634 }
1635
1636 decimal attitudeErrorMean = DECIMAL(attitudeErrorSum) / numCorrect;
1637 decimal fractionCorrect = DECIMAL(numCorrect) / expected.size();
1638 decimal fractionIncorrect = DECIMAL(numIncorrect) / expected.size();
1639
1640 os << "attitude_error_mean " << attitudeErrorMean << std::endl;
1641 os << "attitude_availability " << fractionCorrect << std::endl;
1642 os << "attitude_error_rate " << fractionIncorrect << std::endl;
1643}

1645static void PrintTimeStats(std::ostream &os, const std::string &prefix, const std::vector<long long> &times) {
1646 assert(times.size() > 0);
1647
1648 // print average, min, max, and 95% max
1649 long long sum = 0;
1650 long long min = LONG_MAX;
1651 long long max = 0;
1652 for (int i = 0; i < (int)times.size(); i++) {
1653 assert(times[i] > 0);
1654 sum += times[i];
1655 min = std::min(min, times[i]);
1656 max = std::max(max, times[i]);
1657 }
1658 long average = sum / times.size();
1659 std::vector<long long> sortedTimes = times;
1660 std::sort(sortedTimes.begin(), sortedTimes.end());
1661 // what really is the 95th percentile? Being conservative, we want to pick a value that at least
1662 // 95% of the times are less than. This means: (1) finding the number of times, (2) Finding
1663 // Math.ceil(0.95 * numTimes), and (3) subtracting 1 to get the index.
1664 int ninetyFiveIndex = (int)std::ceil(0.95 * times.size()) - 1;
1665 assert(ninetyFiveIndex >= 0);
1666 long long ninetyFifthPercentile = sortedTimes[ninetyFiveIndex];
1667
1668 os << prefix << "_average_ns " << average << std::endl;
1669 os << prefix << "_min_ns " << min << std::endl;
1670 os << prefix << "_max_ns " << max << std::endl;
1671 os << prefix << "_95%_ns " << ninetyFifthPercentile << std::endl;
1672}

1774void PipelineComparison(const PipelineInputList &expected,
1775 const std::vector<PipelineOutput> &actual,
1776 const PipelineOptions &values) {
1777 if (actual.size() == 0) {
1778 std::cerr << "ERROR: No output! Did you specify any input images? Try --png or --generate." << std::endl;
1779 exit(1);
1780 }
1781
1782 assert(expected.size() == actual.size() && expected.size() > 0);
1783
1784 // TODO: Remove the asserts and print out more reasonable error messages.
1785
1786#define LOST_PIPELINE_COMPARE(precondition, errmsg, comparator, path, isBinary) do { \
1787 if (precondition) { \
1788 UserSpecifiedOutputStream pos(path, isBinary); \
1789 comparator(pos.Stream(), expected, actual, values); \
1790 } else { \
1791 std::cerr << "ERROR: Comparator not applicable: " << errmsg << std::endl; \
1792 exit(1); \
1793 } \
1794 } while (0)
1795
1796 if (values.plotRawInput != "") {
1797 LOST_PIPELINE_COMPARE(expected[0]->InputImage() && expected.size() == 1,
1798 "--plot-raw-input requires exactly 1 input image, but " + std::to_string(expected.size()) + " many were provided.",
1799 PipelineComparatorPlotRawInput, values.plotRawInput, true);
1800 }
1801
1802 if (values.plotInput != "") {
1803 LOST_PIPELINE_COMPARE(expected[0]->InputImage() && expected.size() == 1 && expected[0]->InputStars(),
1804 "--plot-input requires exactly 1 input image, and for centroids to be available on that input image. " + std::to_string(expected.size()) + " many input images were provided.",
1805 PipelineComparatorPlotInput, values.plotInput, true);
1806 }
1807 if (values.plotExpected != "") {
1808 LOST_PIPELINE_COMPARE(expected[0]->InputImage() && expected.size() == 1 && expected[0]->ExpectedStars(),
1809 "--plot-expected-input requires exactly 1 input image, and for expected centroids to be available on that input image. " + std::to_string(expected.size()) + " many input images were provided.",
1810 PipelineComparatorPlotExpected, values.plotExpected, true);
1811 }
1812 if (values.plotOutput != "") {
1813 LOST_PIPELINE_COMPARE(actual.size() == 1 && (actual[0].stars || actual[0].starIds),
1814 "--plot-output requires exactly 1 output image, and for either centroids or star IDs to be available on that output image. " + std::to_string(actual.size()) + " many output images were provided.",
1815 PipelineComparatorPlotOutput, values.plotOutput, true);
1816 }
1817 if (values.printExpectedCentroids != "") {
1818 LOST_PIPELINE_COMPARE(expected[0]->ExpectedStars(),
1819 "--print-expected-centroids requires at least 1 input with expected centroids. " + std::to_string(expected.size()) + " many input images were provided.",
1820 PipelineComparatorPrintExpectedCentroids, values.printExpectedCentroids, false);
1821 }
1822 if (values.printInputCentroids != "") {
1823 LOST_PIPELINE_COMPARE(expected[0]->InputStars(),
1824 "--print-input-centroids requires at least 1 input with centroids. " + std::to_string(expected.size()) + " many input images were provided.",
1825 PipelineComparatorPrintInputCentroids, values.printInputCentroids, false);
1826 }
1827 if (values.printActualCentroids != "") {
1828 LOST_PIPELINE_COMPARE(actual[0].stars,
1829 "--print-actual-centroids requires at least 1 output image, and for centroids to be available on the output images. " + std::to_string(actual.size()) + " many output images were provided.",
1830 PipelineComparatorPrintActualCentroids, values.printActualCentroids, false);
1831 }
1832 if (values.plotCentroidIndices != "") {
1833 LOST_PIPELINE_COMPARE(expected.size() == 1 && expected[0]->InputImage(),
1834 "--plot-centroid-indices requires exactly 1 input with image. " + std::to_string(expected.size()) + " many inputs were provided.",
1835 PipelineComparatorPlotCentroidIndices, values.plotCentroidIndices, true);
1836 }
1837 if (values.compareCentroids != "") {
1838 LOST_PIPELINE_COMPARE(actual[0].stars && expected[0]->ExpectedStars() && values.centroidCompareThreshold,
1839 "--compare-centroids requires at least 1 output image, and for expected centroids to be available on the input image. " + std::to_string(actual.size()) + " many output images were provided.",
