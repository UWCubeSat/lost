65void SerializeKVectorIndex(SerializeContext *ser, const std::vector<decimal> &values, decimal min, decimal max, long numBins) {
66 std::vector<int32_t> kVector(numBins+1); // We store sums before and after each bin
67 decimal binWidth = (max - min) / numBins;
68
69 // generate the k-vector part
70 // Idea: When we find the first star that's across any bin boundary, we want to update all the newly sealed bins
71 long lastBin = 0; // first bin the last star belonged to
72 for (int32_t i = 0; i < (int32_t)values.size(); i++) {
73 if (i > 0) {
74 assert(values[i] >= values[i-1]);
75 }
76 assert(values[i] >= min);
77 assert(values[i] <= max);
78 long thisBin = (long)ceil((values[i] - min) / binWidth); // first bin we belong to
79 assert(thisBin >= 0);
80 assert(thisBin <= numBins); // thisBin == numBins is acceptable since kvector length == numBins + 1
81 // if thisBin > lastBin, then no more stars can be added to those bins between thisBin and lastBin, so set them.
82 for (long bin = lastBin; bin < thisBin; bin++) {
83 kVector[bin] = i; // our index is the number of pairs with shorter distance
84 }
85 lastBin = thisBin;
86 }
87 for (long bin = lastBin; bin <= numBins; bin++) {
88 kVector[bin] = values.size();
89 }
90
91 // verify kvector
92 int32_t lastBinVal = -1;
93 for (const int32_t &bin : kVector) {
94 assert(bin >= lastBinVal);
95 lastBinVal = bin;
96 }
97
98 // metadata fields
99 SerializePrimitive<int32_t>(ser, values.size());
100 SerializePrimitive<decimal>(ser, min);
101 SerializePrimitive<decimal>(ser, max);
102 SerializePrimitive<int32_t>(ser, numBins);
103
104 // kvector index field
105 for (const int32_t &bin : kVector) {
106 SerializePrimitive<int32_t>(ser, bin);
107 }
108}

130long KVectorIndex::QueryLiberal(decimal minQueryDistance, decimal maxQueryDistance, long *upperIndex) const {
131 assert(maxQueryDistance > minQueryDistance);
132 if (maxQueryDistance >= max) {
133 maxQueryDistance = max - DECIMAL(0.00001); // TODO: better way to avoid hitting the bottom bin
134 }
135 if (minQueryDistance <= min) {
136 minQueryDistance = min + DECIMAL(0.00001);
137 }
138 if (minQueryDistance > max || maxQueryDistance < min) {
139 *upperIndex = 0;
140 return 0;
141 }
142 long lowerBin = BinFor(minQueryDistance);
143 long upperBin = BinFor(maxQueryDistance);
144 assert(upperBin >= lowerBin);
145 assert(upperBin <= numBins);
146 // bins[lowerBin-1]=number of pairs <= r < query distance, so it is the index of the
147 // first possible item that might be equal to the query distance
148 int lowerIndex = bins[lowerBin-1];
149 if (lowerIndex >= numValues) {
150 // all pairs have distance less than queried. Return value is irrelevant as long as
151 // numReturned=0
152 return 0;
153 }
154 // bins[upperBin]=number of pairs <= r >= query distance
155 *upperIndex = bins[upperBin];
156 return lowerIndex;
157}

176std::vector<KVectorPair> CatalogToPairDistances(const Catalog &catalog, decimal minDistance, decimal maxDistance) {

221std::vector<uint16_t> TetraPreprocessCatalog(const Catalog& catalog,
222 const decimal maxFovDeg,
223 int numPatternStarsPerFov) {
224 // To eliminate ambiguity (double stars), enforce that all
225 // pattern stars must be > 0.05 degrees apart
226 const decimal starMinSepRad = DegToRad(DECIMAL(0.05));
227 const decimal maxFovRad = DegToRad(maxFovDeg);
228
229 // Catalog is sorted brightest first so brighter stars get priority,
230 // banking on the assumption that brighter stars have lower centroiding error
231 std::vector<uint16_t> patternStarInds{0};
232
233 for (size_t i = 1; i < catalog.size(); i++) {
234 const Vec3& vec = catalog[i].spatial;
235
236 // Reject catalog star if it is a double star of an already-kept star, or if there are
237 // already numPatternStarsPerFov kept stars within maxFov/2 of it
238 bool keep = true;
239 int numPatternStarsInFov = 0;
240 for (uint16_t j : patternStarInds) {
241 decimal angle = Angle(vec, catalog[j].spatial);
242 if (angle < starMinSepRad) {
243 keep = false;
244 break;
245 }
246 if (angle < maxFovRad / 2 && ++numPatternStarsInFov >= numPatternStarsPerFov) {
247 keep = false;
248 break;
249 }
250 }
251
252 if (keep) {
253 patternStarInds.push_back(i);
254 }
255 }
256 return patternStarInds;
257}

325const int16_t *PairDistanceKVectorDatabase::FindPairsExact(const Catalog &catalog,
326 decimal minQueryDistance, decimal maxQueryDistance, const int16_t **end) const {
327
328 // Instead of computing the angle for every pair in the database, we pre-compute the /cosines/
329 // of the min and max query distances so that we can compare against dot products directly! As
330 // angle increases, cosine decreases, up to DECIMAL_M_PI (and queries larger than that don't really make
331 // sense anyway)
332 assert(maxQueryDistance <= DECIMAL_M_PI);
333
334 decimal maxQueryCos = DECIMAL_COS(minQueryDistance);
335 decimal minQueryCos = DECIMAL_COS(maxQueryDistance);
336
337 long liberalUpperIndex;
338 long liberalLowerIndex = index.QueryLiberal(minQueryDistance, maxQueryDistance, &liberalUpperIndex);
339 // now we need to find the first and last index that actually matches the query
340 // step the lower index forward
341 // There's no good reason to be using >= and <= for the comparison against max/min, but the tests fail otherwise (because they use angle, with its acos, instead of forward cos like us). It's an insignificant difference.
342 while (liberalLowerIndex < liberalUpperIndex
343 && catalog[pairs[liberalLowerIndex*2]].spatial * catalog[pairs[liberalLowerIndex*2+1]].spatial >= maxQueryCos
344 )
345 { liberalLowerIndex++; }
346
347 // step the upper index backward
348 while (liberalLowerIndex < liberalUpperIndex
349 // the liberalUpperIndex is past the end of the logically returned range, so we need to subtract 1
350 && catalog[pairs[(liberalUpperIndex-1)*2]].spatial * catalog[pairs[(liberalUpperIndex-1)*2+1]].spatial <= minQueryCos
351 )
352 { liberalUpperIndex--; }
353
354 *end = &pairs[liberalUpperIndex * 2];
355 return &pairs[liberalLowerIndex * 2];
356}

374void SerializeTetraDatabase(SerializeContext* ser,
375 const Catalog& catalog,
376 decimal maxFovDeg,
377 const std::vector<uint16_t>& catalogPatternStarInds,
378 int numEdgeRatioBins) {
379 const decimal maxFovRad = DegToRad(maxFovDeg);
380 const decimal cosMaxFov = DECIMAL_COS(maxFovRad);
381
382 const size_t numPatternStars = catalogPatternStarInds.size();
383
384 // Helper to look up a star's spatial vector given Tetra-internal star catalog index
385 auto spatial = [&](size_t tetraInd) -> const Vec3& {
386 return catalog[catalogPatternStarInds[tetraInd]].spatial;
387 };
388
389 // Every star is a pattern star. Each star is marked unavailable after being
390 // used as the first vertex, so each 4-star group is generated exactly once.
391 std::vector<bool> available(numPatternStars, true);
392
393 std::vector<TetraPattern> patterns;
394
395 for (size_t firstStarInd = 0; firstStarInd < numPatternStars; firstStarInd++) {
396 available[firstStarInd] = false;
397
398 const Vec3& v = spatial(firstStarInd);
399
400 // Find all remaining pattern stars within maxFov
401 std::vector<uint16_t> nearbyStars;
402 for (size_t starId = 0; starId < numPatternStars; starId++) {
403 if (!available[starId]) continue;
404 if (v * spatial(starId) > cosMaxFov) {
405 nearbyStars.push_back(starId);
406 }
407 }
408 CombinationIterator combos(3, nearbyStars.size());
409 while (combos.next()) {
410 const std::vector<int>& idx = combos.indices();
411 TetraPattern patt = {(uint16_t)firstStarInd,
412 nearbyStars[idx[0]],
413 nearbyStars[idx[1]],
414 nearbyStars[idx[2]]};
415
416 bool pattFits = AllPairsWithinFov(
417 TetraConstants::numPatternStars,
418 [&](int i) -> const Vec3& { return spatial(patt[i]); },
419 cosMaxFov);
420
421 if (pattFits) {
422 patterns.push_back(patt);
423 }
424 }
425 }
426
427 std::cerr << "Tetra found " << patterns.size() << " patterns" << std::endl;
428 // Ensure load factor just < 0.5 to prevent against cycling
429 size_t patternTableSize = 2 * patterns.size() + 1;
430 std::vector<TetraPattern> patternTable(patternTableSize);
431 for (const TetraPattern& patt : patterns) {
432 std::vector<decimal> edgeRatios = TetraEdgeRatios(
433 TetraConstants::numPatternStars, [&](int i) -> const Vec3& { return spatial(patt[i]); });
434
435 std::vector<int> binnedEdgeRatios;
436 for (decimal edgeRatio : edgeRatios) {
437 binnedEdgeRatios.push_back(int(edgeRatio * numEdgeRatioBins));
438 }
439
440 size_t patternInd =
441 TetraFindPatternIndex(binnedEdgeRatios.data(), binnedEdgeRatios.size(), numEdgeRatioBins, patternTableSize);
442 for (size_t probeStep = 0; probeStep < patternTableSize; probeStep++) {
443 size_t index = (patternInd + probeStep * probeStep) % patternTableSize;
444 if (patternTable[index][0] == 0 && patternTable[index][1] == 0) {
445 patternTable[index] = patt;
446 break;
447 }
448 }
449 }
450 SerializePrimitive<decimal>(ser, maxFovDeg); // maxFov_
451 SerializePrimitive<int32_t>(ser, numEdgeRatioBins); // numEdgeRatioBins_
452 SerializePrimitive<uint64_t>(ser, patternTable.size()); // patternTableSize_
453 SerializePrimitive<uint16_t>(ser, catalogPatternStarInds.size()); // tetraCatalogSize_
454 for (const TetraPattern& patt : patternTable) {
455 for (int i = 0; i < TetraConstants::numPatternStars; i++) {
456 SerializePrimitive<uint16_t>(ser, patt[i]);
457 }
458 }
459 for (uint16_t ind : catalogPatternStarInds) {
460 SerializePrimitive<uint16_t>(ser, ind);
461 }
462}

484const unsigned char *MultiDatabase::SubDatabasePointer(int32_t magicValue) const {
485 DeserializeContext desValue(buffer);
486 DeserializeContext *des = &desValue; // just for naming consistency with how we use `des` elsewhere
487
488 assert(magicValue != 0);
489 while (true) {
490 int32_t curMagicValue = DeserializePrimitive<int32_t>(des);
491 if (curMagicValue == 0) {
492 return nullptr;
493 }
494 uint32_t dbFlags = DeserializePrimitive<uint32_t>(des);
495
496 // Ensure that our database is using the same type as the runtime.
497 #ifdef LOST_FLOAT_MODE
498 if (!isFlagSet(dbFlags, MULTI_DB_FLOAT_FLAG)) {
499 std::cerr << "LOST was compiled in float mode. This database was serialized in double mode and is incompatible." << std::endl;
500 exit(1);
501 }
502 #else
503 if (isFlagSet(dbFlags, MULTI_DB_FLOAT_FLAG)) {
504 std::cerr << "LOST was compiled in double mode. This database was serialized in float mode and is incompatible." << std::endl;
505 exit(1);
506 }
507 #endif
508
509 uint32_t dbLength = DeserializePrimitive<uint32_t>(des);
510 assert(dbLength > 0);
511 DeserializePadding<uint64_t>(des); // align to an 8-byte boundary
512 const unsigned char *curSubDatabasePointer = DeserializeArray<unsigned char>(des, dbLength);
513 if (curMagicValue == magicValue) {
514 return curSubDatabasePointer;
515 }
516 }
517 // shouldn't ever make it here. Compiler should remove this assertion as unreachable.
518 assert(false);
519}
